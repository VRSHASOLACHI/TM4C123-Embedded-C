#include "tm4c123gh6pm.h"
#include "PLL.h"
#include <stdint.h>

void PortA_Init(void)
{
	unsigned long volatile delay;
	/*PORT Initialization*/
	SYSCTL_RCGC2_R |= 0x01; // Port A clock
	delay = SYSCTL_RCGC2_R; // wait 3-5 bus cycles
	GPIO_PORTA_DIR_R |= 0x20; // PA5 output
	GPIO_PORTA_AFSEL_R &= ~0x20; // not alternative
	GPIO_PORTA_AMSEL_R &= ~0x20; // no analog
	GPIO_PORTA_PCTL_R &= ~0x00F00000; // bits for PA5
	GPIO_PORTA_DEN_R |= 0x20; // enable PA5
}
void SysInit(void)
{
	NVIC_ST_CTRL_R = 0;
	NVIC_ST_CURRENT_R = 0;// any write to current clears it
	//NVIC_SYS_PRI3_R = NVIC_SYS_PRI3_R&0x00FFFFFF;// priority 0
	NVIC_ST_CTRL_R = 0x00000005;// enable with core clock and interrupts
}
void SysLoad(unsigned long period)
{
	NVIC_ST_RELOAD_R = period -1;
	NVIC_ST_CURRENT_R = 0;       // any value written to CURRENT clears  
	while((NVIC_ST_CTRL_R&0x00010000)==0){ // wait for count flag 
		}
}

int main(void)
{
	PLL_Init();
	PortA_Init();
	SysInit();
	while(1)
	{
		unsigned int i;
		for (i = 56000; i <= 120000; i = i + 1000)
		{
		GPIO_PORTA_DATA_R = 0x20;// PA5 is high
		SysLoad(i);
		GPIO_PORTA_DATA_R = 0x00;// Pins are low
		SysLoad(1600000-i);
		}
		
		for (i = 120000; i >= 56000 ; i = i - 1000)
		{
		GPIO_PORTA_DATA_R = 0x20;
		SysLoad(i);
		GPIO_PORTA_DATA_R = 0x00;
		SysLoad(1600000-i);
		}
	}
}
